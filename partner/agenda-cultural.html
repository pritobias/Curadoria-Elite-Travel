#!/usr/bin/env python3 # Agenda Cultural CET — teste independente de scraping (sem APIs) # Dependências: pip install requests beautifulsoup4 # Exemplo: python agenda-cultural-scraping-teste.py --cidade "São Paulo" --data 2026-11-20 --categoria Teatro # NÃO publica resultados, NÃO modifica o site e NÃO garante cobertura de todas as bilheterias. import argparse import json import re import sys import time from datetime import datetime from urllib.parse import quote, urlparse from xml.etree import ElementTree import requests from bs4 import BeautifulSoup HEADERS = {"User-Agent": "Mozilla/5.0 (compatible; CETAgendaResearchTest/0.1)"} TIMEOUT = 12 MAX_LINKS = 15 def request_text(url): """Acessa somente HTTP(S) público; respeita erros e bloqueios do site.""" if urlparse(url).scheme not in ("http", "https"): return None try: response = requests.get(url, headers=HEADERS, timeout=TIMEOUT, allow_redirects=True) response.raise_for_status() if len(response.content) > 3_000_000: return None return response.text except requests.RequestException: return None def search_public_pages(city, date, category): """Busca links em RSS público do Bing; não usa API nem lista de eventos.""" terms = f'"{city}" "{date}" {category} evento ingressos programação' url = 'https://www.bing.com/search?format=rss&q=' + quote(terms) xml = request_text(url) if not xml: return [] try: root = ElementTree.fromstring(xml) except ElementTree.ParseError: return [] links = [] for item in root.findall('.//item'): link = item.findtext('link', '').strip() if link.startswith('https://') and link not in links: links.append(link) return links[:MAX_LINKS] def flatten_jsonld(value): if isinstance(value, list): for child in value: yield from flatten_jsonld(child) elif isinstance(value, dict): if '@graph' in value: yield from flatten_jsonld(value['@graph']) yield value def is_event(node): kind = node.get('@type', []) if isinstance(kind, str): kind = [kind] return any(str(x).lower().endswith('event') for x in kind) def text_value(value): if isinstance(value, dict): return str(value.get('name') or value.get('value') or '').strip() return str(value or '').strip() def extract_address(location): if isinstance(location, list): location = location[0] if location else {} if not isinstance(location, dict): return '', '' venue = text_value(location.get('name')) address = location.get('address') or {} if isinstance(address, str): return venue, address if not isinstance(address, dict): return venue, '' parts = [address.get(k) for k in ('streetAddress', 'addressLocality', 'addressRegion', 'postalCode', 'addressCountry')] return venue, ', '.join(str(p) for p in parts if p) def extract_offer(offers): if isinstance(offers, list): offers = offers[0] if offers else {} if not isinstance(offers, dict): return None, None, None price = offers.get('price') currency = offers.get('priceCurrency') availability = offers.get('availability') if price is None or not currency: price_label = None else: price_label = f'{price} {currency}' return price_label, availability, offers.get('url') def parse_events(html, page_url): soup = BeautifulSoup(html, 'html.parser') found = [] for script in soup.select('script[type="application/ld+json"]'): try: payload = json.loads(script.string or script.get_text()) except (json.JSONDecodeError, TypeError): continue for node in flatten_jsonld(payload): if not is_event(node): continue title = text_value(node.get('name')) start = text_value(node.get('startDate')) if not title or not start: continue venue, address = extract_address(node.get('location')) price, availability, offer_url = extract_offer(node.get('offers')) image = node.get('image') if isinstance(image, list): image = image[0] if image else None if isinstance(image, dict): image = image.get('url') ticket_url = offer_url if isinstance(offer_url, str) and offer_url.startswith('https://') else None found.append({ 'titulo': title, 'inicio': start, 'local': venue or None, 'endereco': address or None, 'sinopse': BeautifulSoup(text_value(node.get('description')), 'html.parser').get_text(' ', strip=True) or None, 'imagem': image if isinstance(image, str) and image.startswith('https://') else None, 'preco_confirmado_na_pagina': price, 'disponibilidade_informada_na_pagina': availability, 'link_de_ingressos_informado_na_pagina': ticket_url, 'fonte': page_url, 'aviso': 'Dados extraídos da página pública; preço e disponibilidade exigem conferência na bilheteria.' }) return found def matches_date(event, date): return event['inicio'][:10] == date def matches_city(event, city): import unicodedata def normalize(s): return ''.join(c for c in unicodedata.normalize('NFKD', s.lower()) if not unicodedata.combining(c)) place = f"{event.get('local') or ''} {event.get('endereco') or ''}" return normalize(city) in normalize(place) def main(): parser = argparse.ArgumentParser(description='Teste de scraping da Agenda Cultural CET, sem API.') parser.add_argument('--cidade', required=True) parser.add_argument('--data', required=True, help='AAAA-MM-DD') parser.add_argument('--categoria', default='eventos') args = parser.parse_args() try: datetime.strptime(args.data, '%Y-%m-%d') except ValueError: parser.error('Data inválida. Use AAAA-MM-DD.') links = search_public_pages(args.cidade, args.data, args.categoria) events = [] seen = set() for link in links: html = request_text(link) if not html: continue for event in parse_events(html, link): if not matches_date(event, args.data) or not matches_city(event, args.cidade): continue key = (event['titulo'].casefold(), event['inicio'], event.get('local')) if key not in seen: seen.add(key) events.append(event) if len(events) >= 5: break time.sleep(0.5) print(json.dumps({ 'cidade': args.cidade, 'data': args.data, 'categoria_pesquisada': args.categoria, 'paginas_encontradas_na_busca': len(links), 'eventos_extraidos': events[:5], 'observacao': ('Nenhum resultado não significa ausência de eventos. ' 'Sites podem bloquear coleta ou não publicar dados estruturados. ' 'A categoria orienta a busca, mas não é validada automaticamente.') }, ensure_ascii=False, indent=2)) if __name__ == '__main__': main()
